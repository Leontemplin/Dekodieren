    <script>
        let currentActiveId = null;

        // Sicherer Speicher-Abruf
        function getStorageData() {
            try {
                const data = localStorage.getItem('decodedTexts');
                return data ? JSON.parse(data) : [];
            } catch (e) {
                console.error(e);
                return [];
            }
        }

        // Hauptfunktion zum Zerlegen
        function processText(customText = null) {
            try {
                const text = customText !== null ? customText : document.getElementById('inputText').value.trim();
                if (customText === null) {
                    document.getElementById('inputText').value = text;
                    currentActiveId = null;
                }
                
                const workspace = document.getElementById('workspace');
                workspace.innerHTML = ''; 

                if (!text) {
                    workspace.innerHTML = '<p class="placeholder-text">Bitte zuerst einen englischen Text eingeben!</p>';
                    return;
                }

                const words = text.split(/\s+/);

                words.forEach((word, index) => {
                    if (word.length === 0) return;

                    const cleanWord = word.replace(/[.,\/#!$%\^&\*;:{}=\-_`~()?]/g,"");
                    const wordBlock = document.createElement('div');
                    wordBlock.className = "word-block";

                    const inputId = `input-word-${index}`;

                    wordBlock.innerHTML = `
                        <div class="word-header">
                            <span class="eng-word">${word}</span>
                            <button onclick="translateSingleWord('${cleanWord}', '${inputId}')" class="btn-lookup">🔍</button>
                        </div>
                        <input type="text" 
                               id="${inputId}"
                               class="de-input" 
                               placeholder="______" 
                               oninput="autoSaveTranslations()">
                    `;
                    workspace.appendChild(wordBlock);
                });
            } catch (err) {
                alert("Fehler beim Zerlegen: " + err.message);
            }
        }

        // Funktion zum Speichern
        function saveCurrentText() {
            try {
                const text = document.getElementById('inputText').value.trim();
                if (!text) return alert("Bitte zuerst einen Text eingeben!");

                let savedTexts = getStorageData();
                
                if (savedTexts.some(item => item.original === text)) {
                    return alert("Dieser Text existiert bereits in deiner Liste!");
                }

                const newItem = {
                    id: Date.now().toString(),
                    original: text,
                    translations: {}
                };

                savedTexts.unshift(newItem);
                localStorage.setItem('decodedTexts', JSON.stringify(savedTexts));
                
                currentActiveId = newItem.id;
                processText(text);
                loadSavedTexts();
            } catch (err) {
                alert("Fehler beim Speichern: " + err.message);
            }
        }

        // Funktion zum Laden der Liste

                const savedList = document.getElementById('savedList');
                if (!savedList) return;

                const savedTexts = getStorageData();

                    return;
                }

                savedList.innerHTML = '';
                savedTexts.forEach(item => {
                    const div = document.createElement('div');
                    div.className = "saved-item";
                    
                    const shortText = item.original.length > 35 ? item.original.substring(0, 35) + '...' : item.original;

                    div.innerHTML = `
